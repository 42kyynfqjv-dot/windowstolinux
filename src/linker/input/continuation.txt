// Native compiled-block control. No guest byte decoding or host-stack recursion.
enum Flow { Next(u32), Call{target:u32,next:u32}, Return, Terminal(u32) }
struct Activation { next:u32, stack_len:usize, saved:[u32;3] }
fn execute(entry:u32,s:&mut Machine)->u32 {
 let mut pc=entry;
 let mut activations:Vec<Activation>=Vec::new();
 loop { match block(pc,s) {
  Flow::Next(next)=>pc=next,
  Flow::Call{target,next}=>{
   // Same four-active-call bound as the production ownership proof.
   if activations.len()>=MAX_ACTIVATIONS {return 203;}
   activations.push(Activation{next,stack_len:s.stack.len(),saved:[s.r[3],s.r[4],s.r[5]]});
   s.stack.push(next);pc=target;
  },
  Flow::Return=>match activations.last(){
   Some(a)=>{
    // Validate before removing either the activation or its protected word.
    if s.stack.len()!=a.stack_len+1 || s.stack.last()!=Some(&a.next)
       || s.r[3..]!=a.saved {return 203;}
    pc=a.next;s.stack.pop();activations.pop();
   },
   None=>return s.r[0],
  },
  // Host acquisition failure must bypass all abandoned ordinary continuations.
  // This terminal effect is not a guest exception or Windows unwind mapping.
  Flow::Terminal(code)=>return code,
 }}
}
